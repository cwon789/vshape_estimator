cmake_minimum_required(VERSION 3.16)
project(ros2_bot_estimator_lidar_vshape VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
option(VSHAPE_BUILD_ROS "Build ROS 2 node and install ament package" ON)
option(VSHAPE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(VSHAPE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Boost REQUIRED CONFIG)
# Match the Eigen version used by the installed ROS libraries.
set_property(TARGET Eigen3::Eigen PROPERTY INTERFACE_INCLUDE_DIRECTORIES /usr/include/eigen3)
add_library(vshape_core src/filter.cpp src/detector.cpp src/tracker.cpp)
target_include_directories(vshape_core PUBLIC include)
target_link_libraries(vshape_core PUBLIC Eigen3::Eigen Boost::boost)
add_executable(vshape_replay src/replay_main.cpp)
target_link_libraries(vshape_replay PRIVATE vshape_core)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  foreach(name models filter detector tracker output)
    add_executable(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE vshape_core GTest::gtest_main)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  add_executable(vshape_consistency test/test_consistency.cpp)
  target_link_libraries(vshape_consistency PRIVATE vshape_core)
  add_test(NAME consistency COMMAND vshape_consistency)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME validation_contract COMMAND ${Python3_EXECUTABLE} -m unittest discover -s test -p test_validation.py)
  set_tests_properties(validation_contract PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()

if(VSHAPE_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(diagnostic_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  add_executable(vshape_node src/node.cpp)
  target_link_libraries(vshape_node vshape_core)
  ament_target_dependencies(vshape_node rclcpp rcl_interfaces sensor_msgs nav_msgs geometry_msgs diagnostic_msgs std_srvs)
  install(TARGETS vshape_node DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
